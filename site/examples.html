Counter (GUI)

Download counter.vxl 
create a window titled "Counter"

count = 0
label = add a text titled "Count: 0"
button = add a button titled "Increase"

button.on click {
    count = count + 1
    label.text = "Count: " + count
}

window.show

Timer

create a window titled "Timer"

label = add a text titled "waiting..."

after 3 seconds {
    label.text = "Done!"
}

every 1 second {
    print "tick"
}

window.show

Text editor (mini)

create a window titled "Notes"

box = add an output box
box.width = 500
box.height = 300

save = add a button titled "Save"

save.on click {
    json write box.text to "notes.json"
    show a message "Saved!"
}

window.show
